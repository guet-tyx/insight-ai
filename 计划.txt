from browser_use import Agent, Browser, BrowserConfig

class BrowserSessionManager:
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            # 配置 Playwright 浏览器，包含防检测参数与代理
            cls._instance.config = BrowserConfig(
                headless=True,
                disable_security=True,
                extra_chromium_args=["--no-sandbox", "--disable-setuid-sandbox"]
            )
            cls._instance.browser = Browser(config=cls._instance.config)
        return cls._instance

    async def execute_task(self, task_instruction: str, llm_model) -> str:
        # 创建隔离的上下文，避免会话间数据污染
        context = await self.browser.new_context()
        try:
            agent = Agent(
                task=task_instruction,
                llm=llm_model,
                browser_context=context
            )
            result = await agent.run()
            return result
        finally:
            await context.close()

全链路可观测性与 Trace 监控为了解决大模型系统中因非确定性引起的死循环、工具调用失败及 Token 资源浪费问题，系统深度集成 LangSmith 可观测性平台。每次用户请求均会被赋予唯一的 trace_id。LangSmith 实时记录从 SupervisorAgent 分发路由，到各专家 Agent 子图内部推理，再到底层 MCP 工具调用的完整 Trace 树。系统对每个节点的输入 Prompt、输出 Content、温度参数、执行延迟及 Token 消耗量进行持续监控。同时，系统设定了状态漂移熔断规则：若某个 Agent 在单个会话中连续调用相同工具超过 3 次，或状态图迭代步骤超过 20 步，系统将抛出异常并自动终止执行，防止产生异常开销。针对长文档分析场景，系统在 Analyst Agent 前置了上下文动态压缩节点，通过预判 Token 计算并在必要时触发摘要 Reducer，将上下文体积控制在最优质的区间内。结论与工程价值评估Insight AI 平台的工程实施方案通过严谨的技术选型与模块化设计，展示了现代化 AI Agent 系统的工程化落地路径。项目将 LangGraph 的强类型确定性状态机与 FastMCP 的开放式插件扩展标准相融合，有效解决了复杂情报分析场景下的上下文丢失、智能体失控及工具解耦困难等行业典型工程痛点。在技术能力评估与求职匹配度层面，该平台体现了以下核心工程优势：系统架构层面，平台展现了多智能体层次化编排、子图隔离、状态机持久化以及人机协作（HITL）断点恢复等高阶 Agent 工程能力；算法与数据层面，平台实现了 GraphRAG 密集向量与拓扑图谱的复合检索，利用 RRF 算法提升了复杂回答的精准度；微服务与后端层面，平台完成了基于 FastMCP 2.0 的工具服务化改造，具备云原生部署、CI/CD 自动化流水线及全链路 Trace 可观测性。这份精细化的十二周实施计划，不仅为项目开发提供了明确的技术蓝图，也构成了能够展示复合技术实力的生产级工程作品集。
